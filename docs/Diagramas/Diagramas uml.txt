@startuml
!theme plain
skinparam classAttributeIconSize 0

package "Diagrama de Clases" {
class Usuario {
    - id: int
    - nombre: string
    - apellido: string
    - usuario: string
    - correo: string
    - contrasena_hash: string
    - creado_el: DateTime
    + iniciarSesion()
    + registrarse()
    + recuperarContraseña()
}

class Archivo {
    - id: int
    - remitente_id: int
    - destinatario_id: int
    - asunto: string
    - nombre_original: string
    - nombre_servidor: string
    - url: string
    - fecha_subida: DateTime
    - leido: bool
    + guardarArchivo()
    + descargarArchivo()
}

class UsuarioEnChatDeVoz {
    - id: int
    - id_usuario: int
    - nombre_usuario: string
    - sala: string
    - fecha: DateTime
    + unirseSala()
}

class RegistroDeEntrada {
    - id: int
    - id_usuario: int
    - sala: string
    - fecha: DateTime
    + registrarEntrada()
}

class RegistroDeSalida {
    - id: int
    - id_usuario: int
    - sala: string
    - fecha: DateTime
    + registrarSalida()
}

class RegistroSalasCreadasPorUsuario {
    - id: int
    - id_usuario: int
    - sala: string
    - fecha: DateTime
    + crearSala()
}

' Clases de aplicación
class ClienteUnity {
    + moverse()
    + hablar()
    + enviarArchivo()
    + recibirArchivo()
    + verUsuarios()
}

class ServidorPHP {
    + validarCredenciales()
    + gestionarSubida()
    + gestionarDescarga()
    + gestionarNotificaciones()
    + obtenerUsuarios()
    + usuarioSalaDeVoz()
}

class Vivox {
    + crearCanalDeVoz()
    + gestionarUsuarios()
}
}

' Relaciones
Usuario "1" -- "n" Archivo : remitente >
Usuario "1" -- "n" Archivo : destinatario >
Usuario "1" -- "n" UsuarioEnChatDeVoz : participa >
Usuario "1" -- "n" RegistroDeEntrada : entradas >
Usuario "1" -- "n" RegistroDeSalida : salidas >
Usuario "1" -- "n" RegistroSalasCreadasPorUsuario : crea >

ClienteUnity --> Usuario : gestiona
ClienteUnity --> Vivox : conecta/desconecta
ClienteUnity --> ServidorPHP : comunica

ServidorPHP --> Usuario : autentica
ServidorPHP --> Archivo : gestiona archivos
ServidorPHP --> UsuarioEnChatDeVoz : gestiona salas
ServidorPHP --> RegistroDeEntrada : registra entrada
ServidorPHP --> RegistroDeSalida : registra salida
ServidorPHP --> RegistroSalasCreadasPorUsuario : crea sala

@enduml
